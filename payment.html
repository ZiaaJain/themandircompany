const SPREADSHEET_ID = "1-N-XlBC0haaR5i7O3U1XHGDT9_zNHHUVl4Dz-qsalkI"; const SHEET_NAME = "The Mandir Company Website"; // =============================== // RECEIVE NEW ORDER // =============================== function doPost(e) { try { if (!e || !e.parameter) { throw new Error("No order data was received."); } const spreadsheet = SpreadsheetApp.openById(SPREADSHEET_ID); const sheet = spreadsheet.getSheetByName(SHEET_NAME); if (!sheet) { throw new Error( 'Sheet "' + SHEET_NAME + '" was not found.' ); } // Get order information from the form POST const data = { date: e.parameter.date || "", fullName: e.parameter.fullName || "", phone: e.parameter.phone || "", address: e.parameter.address || "", items: e.parameter.items || "", subtotal: e.parameter.subtotal || "", shipping: e.parameter.shipping || "", total: e.parameter.total || "" }; // =============================== // CREATE ORDER NUMBER // =============================== const lastRow = sheet.getLastRow(); let orderNumber; if (lastRow <= 1) { orderNumber = 1; } else { orderNumber = lastRow; } const orderId = "#" + orderNumber; // =============================== // ADD ORDER TO GOOGLE SHEET // =============================== sheet.appendRow([ orderId, data.date, data.fullName, data.phone, data.address, data.items, data.subtotal, data.shipping, data.total ]); // =============================== // SEND CUSTOMER TO CONFIRMATION // =============================== const confirmationURL = "https://ziaajain.github.io/confirmation.html?orderId=" + encodeURIComponent(orderId); return HtmlService .createHtmlOutput(` Order Confirmed

Order placed successfully!

Your order has been received.

Order ID: ${orderId}

Continue `); } catch (error) { console.error("ORDER ERROR:", error); return HtmlService .createHtmlOutput(` Order Error

Order could not be placed

${String(error)}

Please go back and try again.

`); } } // =============================== // GET ALL ORDERS // =============================== function doGet(e) { try { const spreadsheet = SpreadsheetApp.openById(SPREADSHEET_ID); const sheet = spreadsheet.getSheetByName(SHEET_NAME); if (!sheet) { throw new Error( 'Sheet "' + SHEET_NAME + '" was not found.' ); } const rows = sheet.getDataRange().getValues(); const orders = []; // Skip header row for (let i = 1; i < rows.length; i++) { if ( rows[i][0] && rows[i][0].toString().trim() !== "" ) { orders.push({ orderId: String(rows[i][0]), date: String(rows[i][1] || ""), fullName: String(rows[i][2] || ""), phone: String(rows[i][3] || ""), address: String(rows[i][4] || ""), items: String(rows[i][5] || ""), subtotal: String(rows[i][6] || ""), shipping: String(rows[i][7] || ""), total: String(rows[i][8] || "") }); } } return ContentService .createTextOutput( JSON.stringify(orders) ) .setMimeType( ContentService.MimeType.JSON ); } catch (error) { console.error( "GET ORDERS ERROR:", error ); return ContentService .createTextOutput( JSON.stringify({ result: "error", error: String(error), orders: [] }) ) .setMimeType( ContentService.MimeType.JSON ); } }
