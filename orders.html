async function handleProductUpload(event) { event.preventDefault(); const btn = document.getElementById("upload-btn"); btn.innerText = "Uploading photo & publishing..."; btn.disabled = true; const fileInput = document.getElementById("p-image-file").files[0]; if (!fileInput) { alert("Please select an image file first."); btn.innerText = "Upload & Publish Product 🚀"; btn.disabled = false; return; } const reader = new FileReader(); reader.onload = async function() { try { const payload = new URLSearchParams({ formType: "addProduct", name: document.getElementById("p-name").value, price: document.getElementById("p-price").value, category: document.getElementById("p-category").value, desc: document.getElementById("p-desc").value, fileData: reader.result, mimeType: fileInput.type, fileName: fileInput.name }); await fetch(GOOGLE_SCRIPT_URL, { method: "POST", body: payload }); alert("Product published successfully! It is now live."); document.getElementById("addProductForm").reset(); } catch (error) { alert("Failed to publish product: " + error.message); } finally { btn.innerText = "Upload & Publish Product 🚀"; btn.disabled = false; } }; reader.readAsDataURL(fileInput); }
